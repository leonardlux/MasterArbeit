\subsection{Stabilizer Codes}
\begin{frame}{Stabilizer Codes and Syndromes}
    \begin{itemize}
        \item Stabilizer group: 
            $ \mathcal{S} = \langle S_{(1)},...,S_{(m)}\rangle \subset \mathcal{P}_n$
    \end{itemize}
    \begin{block}{Stabilizer \& Syndrome}
          \begin{equation*}
                 S_{(i)} \ket{\Psi}_L= (-1)^{s^{(i)}} \ket{\Psi}_L
          \end{equation*}
          defines syndrome:
          \begin{equation*}
                \mathbf{s} = \left(s^{(1)}, s^{(2)}, ..., s^{(m)} \right) \in \mathbb{Z}_2^m
          \end{equation*}
    \end{block}
    \begin{itemize}
        \item Codespace:  $\mathcal{H}_{\mathbf{s}_0} \subset \mathcal{H}_{2^n}$
        \item Error subspaces: $\mathcal{H}_\mathbf{s}$
                % = \{ \ket{\Psi}_L \in \mathcal{H} \ |\ g^{(i)}\ket{\Psi} = (-1)^{s_i} \ket{\Psi}_L \}$ \\
    \end{itemize} 
\end{frame}

\section{Syndrome Extraction}
\subsection{Steane-Type Error Correction}

\begin{frame}
    \onslide<1->{
    \begin{block}{Circuit: $Z$-type Half Cycle}
    \begin{center}
    \begin{quantikz}
        \lstick{$\ket{\Psi}_L$}&\qwbundle{n}&\ghost{E_X}&\ctrl{1}&&\\
        \lstick{$\ket{+}_L$}&\qwbundle{n}&\ghost{E_X}&\targ{}&\meter{Z}&\setwiretype{c}
    \end{quantikz}
    \end{center}
    \end{block}
    }
    \onslide<1->{
        \begin{itemize}
            \item Logically trivial:
        \end{itemize}
        \begin{equation*}
            C^{\ket{\Psi}}NOT^{\ket{+}} \left(\ket{\Psi}_L \otimes \ket{+}_L \right)  = \ket{\Psi}_L \otimes \ket{+}_L
        \end{equation*}
    }

    \onslide<2->{
        \begin{itemize}
            \item $X$ error on $i$-th qubit:
        \end{itemize}
        \begin{equation*}
            C^{\ket{\Psi}}NOT^{\ket{+}}  (X_i \otimes I) \left(\ket{\Psi}_L \otimes \ket{+}_L \right)  = X_i\ket{\Psi}_L \otimes X_i\ket{+}_L
        \end{equation*}
    }

\end{frame}

\section{Decoder}
\begin{frame}{Decoder}
    \begin{enumerate}
        \onslide<1->{
        \item Maximum Likelihood Decoding (ML)
        \begin{itemize}
            \item Determine most probable logical class of errors 
            \begin{itemize}
                \item Coset of stabilizer group in the centralizer 
                \item $\Rightarrow$ Maximum likelihood of successful recovery 
            \end{itemize}
            \item Generally $\#P$-hard \cite{MP_sharp_P_hard}
            \item Efficient solvable for certain problems {\cite{efficient_decoding_algorithm}} 
            \begin{itemize}
                \item Surface code
                \item Bit- \& phase-flip noise
            \end{itemize}
        \end{itemize}}
        \onslide<2->{
        \item Minimum Weight Perfect Matching  (MWPM)
        \begin{itemize}
            \item Determines most probable error 
            \begin{itemize}
                \item MWPM on matching graph \cite{pymatching_loger_paper}
            \end{itemize}
            \item Efficient decoding 
            \item Below optimal success rate 
        \end{itemize}}
    \end{enumerate}
\end{frame}

\subsection*{Noise model equations}
\begin{frame}
\begin{block}{$X$-Errors}
    \begin{equation}
        \varepsilon_{X}(\rho) = (1-p) \rho + p \cdot X \rho X
    \end{equation}
\end{block}

% Depo 1
\begin{block}{Depolarizing Errors}
    \begin{equation}
        \varepsilon_{D_1}(\rho) = (1-p) \rho + \frac{p}{3} \sum^3_{i=1} E_i \rho E_i
    \end{equation}
    \begin{equation*}
        E_i \in \{ X, Y, Z \}
    \end{equation*}

    % Depo 2
        \begin{equation}
            \varepsilon_{D_2}(\rho) = (1-p) \rho + \frac{p}{15} \sum_{i=1}^{15} E_i \rho E_i,
        \end{equation}
